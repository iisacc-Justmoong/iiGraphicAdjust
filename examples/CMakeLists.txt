cmake_minimum_required(VERSION 3.24)
# iisdk:consumer-application
project(iiGraphicAdjustExamples LANGUAGES CXX)

# These are consumer applications: each SDK is found and linked independently.
find_package(iiGraphicAdjust 0.6 CONFIG REQUIRED)
find_package(iiFileProviderNative 0.6 CONFIG REQUIRED)
foreach(example IN ITEMS Objects LocalCorrections Develop Image)
    if(example STREQUAL "Objects")
        set(source objects.cpp)
        set(target iiGraphicAdjustObjectsExample)
    elseif(example STREQUAL "LocalCorrections")
        set(source local_corrections.cpp)
        set(target iiGraphicAdjustLocalCorrectionsExample)
    elseif(example STREQUAL "Develop")
        set(source develop.cpp)
        set(target iiGraphicAdjustDevelopExample)
    else()
        set(source adjust.cpp)
        set(target iiGraphicAdjustExample)
    endif()
    add_executable(${target} ${source})
    target_compile_features(${target} PRIVATE cxx_std_23)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_link_libraries(${target} PRIVATE iiGraphicAdjust::iiGraphicAdjust iiFileProvider::Native)
endforeach()
